🚀 Introduction à JavaScript

JavaScript est un langage de programmation qui permet de rendre les pages web interactives. C'est l'un des trois piliers du développement web avec HTML (structure) et CSS (apparence).

Comment utiliser JavaScript ?

Il existe trois façons d'inclure du JavaScript dans une page web :

1. JavaScript inline (dans les balises HTML) : 
<button onclick="alert('Bonjour !')">Cliquez-moi</button>
2. JavaScript interne (dans la balise <script>) : 
<script>
    console.log('Bonjour le monde !');
</script>
3. JavaScript externe (fichier .js séparé) : 
<script src="monscript.js"></script>

📦 Les Variables

Les variables sont des conteneurs qui stockent des données. En JavaScript, on peut déclarer des variables avec let, const ou var.

let - Variable modifiable

let prenom = "Marie";
prenom = "Jean"; // ✅ OK
console.log(prenom); // Affiche: Jean

const - Variable constante (non modifiable)

const age = 25;
age = 30; // ❌ ERREUR
console.log(age);

var - Ancienne méthode (à éviter)

var ville = "Paris";
// Évitez d'utiliser var, préférez let ou const
💡 Astuce : Utilisez const par défaut, et let uniquement si vous devez modifier la valeur.

🎨 Types de Données

1. String (Chaîne de caractères)

let message = "Bonjour";
let nom = 'Claude';
let phrase = `Je m'appelle ${nom}`; // Template literal

2. Number (Nombre)

let entier = 42;
let decimal = 3.14;
let negatif = -7;

3. Boolean (Booléen)

let estVrai = true;
let estFaux = false;

4. Undefined et Null

let nonDefini; // undefined
let vide = null; // null

Vérifier le type

console.log(typeof "Bonjour"); // "string"
console.log(typeof 42); // "number"
console.log(typeof true); // "boolean"

➕ Opérateurs

Opérateurs arithmétiques

Opérateur Description Exemple
+ Addition 5 + 3 = 8
- Soustraction 5 - 3 = 2
* Multiplication 5 * 3 = 15
/ Division 6 / 3 = 2
% Modulo (reste) 5 % 2 = 1
** Puissance 2 ** 3 = 8

Opérateurs de comparaison

5 == "5"   // true (égalité faible)
5 === "5"  // false (égalité stricte)
5 != "5"   // false
5 !== "5"  // true
5 > 3      // true
5 < 3      // false
5 >= 5     // true
5 <= 4     // false
⚠️ Attention : Utilisez toujours === et !== pour éviter les surprises !

🔀 Conditions

if / else

let age = 18;

if (age >= 18) {
    console.log("Vous êtes majeur");
} else {
    console.log("Vous êtes mineur");
}

else if

let note = 15;

if (note >= 16) {
    console.log("Très bien");
} else if (note >= 14) {
    console.log("Bien");
} else if (note >= 10) {
    console.log("Assez bien");
} else {
    console.log("Insuffisant");
}

Opérateur ternaire

let age = 20;
let statut = age >= 18 ? "majeur" : "mineur";
console.log(statut); // "majeur"

switch

let jour = "lundi";

switch (jour) {
    case "lundi":
        console.log("Début de semaine");
        break;
    case "vendredi":
        console.log("Bientôt le weekend !");
        break;
    case "samedi":
    case "dimanche":
        console.log("C'est le weekend !");
        break;
    default:
        console.log("Jour normal");
}

🔄 Boucles

Boucle for

for (let i = 0; i < 5; i++) {
    console.log(i); // Affiche 0, 1, 2, 3, 4
}

Boucle while

let compteur = 0;

while (compteur < 5) {
    console.log(compteur);
    compteur++;
}

Boucle do...while

let nombre = 0;

do {
    console.log(nombre);
    nombre++;
} while (nombre < 5);

for...of (pour les tableaux)

let fruits = ["pomme", "banane", "orange"];

for (let fruit of fruits) {
    console.log(fruit);
}

⚙️ Fonctions

Déclaration classique

function direBonjour(nom) {
    return "Bonjour " + nom;
}

console.log(direBonjour("Marie")); // "Bonjour Marie"

Fonction avec plusieurs paramètres

function additionner(a, b) {
    return a + b;
}

let resultat = additionner(5, 3);
console.log(resultat); // 8

Fonction fléchée (arrow function)

const multiplier = (a, b) => {
    return a * b;
}

// Version courte (si une seule ligne)
const doubler = x => x * 2;

console.log(doubler(5)); // 10

Paramètres par défaut

function saluer(nom = "visiteur") {
    return `Bonjour ${nom} !`;
}

console.log(saluer()); // "Bonjour visiteur !"
console.log(saluer("Paul")); // "Bonjour Paul !"

📋 Tableaux (Arrays)

Créer un tableau

let fruits = ["pomme", "banane", "orange"];
let nombres = [1, 2, 3, 4, 5];
let mixte = ["texte", 42, true];

Accéder aux éléments

console.log(fruits[0]); // "pomme"
console.log(fruits[2]); // "orange"

Méthodes principales

// Ajouter à la fin
fruits.push("fraise");

// Retirer du début
fruits.shift();

// Ajouter au début
fruits.unshift("kiwi");

// Retirer de la fin
fruits.pop();

// Longueur du tableau
console.log(fruits.length);

// Trouver un élément
let index = fruits.indexOf("banane");

Parcourir un tableau

// forEach
fruits.forEach(function(fruit) {
    console.log(fruit);
});

// map (crée un nouveau tableau)
let majuscules = fruits.map(fruit => fruit.toUpperCase());

// filter (filtre les éléments)
let nombres = [1, 2, 3, 4, 5];
let pairs = nombres.filter(n => n % 2 === 0); // [2, 4]

🎁 Objets

Créer un objet

let personne = {
    nom: "Dupont",
    prenom: "Jean",
    age: 30,
    ville: "Paris"
};

Accéder aux propriétés

// Notation point
console.log(personne.nom); // "Dupont"

// Notation crochets
console.log(personne["prenom"]); // "Jean"

Modifier une propriété

personne.age = 31;
personne["ville"] = "Lyon";

Ajouter une propriété

personne.email = "jean@example.com";

Méthodes dans les objets

let voiture = {
    marque: "Renault",
    modele: "Clio",
    annee: 2020,
    demarrer: function() {
        console.log("La voiture démarre !");
    }
};

voiture.demarrer(); // "La voiture démarre !"

🌐 Manipulation du DOM

Le DOM (Document Object Model) permet d'interagir avec les éléments HTML.

Sélectionner des éléments

// Par ID
let element = document.getElementById("monId");

// Par classe
let elements = document.getElementsByClassName("maClasse");

// Par balise
let paragraphes = document.getElementsByTagName("p");

// Sélecteur CSS (le plus utilisé)
let premier = document.querySelector(".maClasse");
let tous = document.querySelectorAll(".maClasse");

Modifier le contenu

let titre = document.querySelector("h1");
titre.textContent = "Nouveau titre";
titre.innerHTML = "<strong>Titre en gras</strong>";

Modifier le style

let boite = document.querySelector(".boite");
boite.style.color = "red";
boite.style.backgroundColor = "yellow";
boite.style.fontSize = "20px";

Ajouter/retirer des classes

element.classList.add("nouvelle-classe");
element.classList.remove("ancienne-classe");
element.classList.toggle("actif"); // Ajoute ou retire

Événements

// Événement au clic
let bouton = document.querySelector("button");
bouton.addEventListener("click", function() {
    alert("Bouton cliqué !");
});

// Événement sur un formulaire
let formulaire = document.querySelector("form");
formulaire.addEventListener("submit", function(e) {
    e.preventDefault(); // Empêche le rechargement
    console.log("Formulaire soumis");
});

Créer et ajouter des éléments

// Créer un élément
let nouveauP = document.createElement("p");
nouveauP.textContent = "Nouveau paragraphe";

// L'ajouter au DOM
document.body.appendChild(nouveauP);
Exemple complet : 
<button id="monBouton">Cliquez-moi</button>
<p id="compteur">0</p>

<script>
let bouton = document.getElementById("monBouton");
let compteur = document.getElementById("compteur");
let compte = 0;

bouton.addEventListener("click", function() {
    compte++;
    compteur.textContent = compte;
});
</script>

🎯 Conseils pour bien débuter

1. Pratiquez régulièrement
La programmation s'apprend en pratiquant. Essayez de coder un peu chaque jour.
2. Utilisez la console
console.log() est votre meilleur ami pour déboguer et comprendre ce qui se passe dans votre code.
3. Lisez les messages d'erreur
Les messages d'erreur vous indiquent où et pourquoi votre code ne fonctionne pas.
4. Commentez votre code
